C_analyze_click

set description "Executes Python code"
___run__
// HACK:
// Strip cursor_roi:
cursor_roi = removeSubstring(cursor_roi, "None")
cursor_roi = removeSubstring(cursor_roi, ";")



// Determine whether to (re)play audio file or not


if (cursor_roi === 'image1') {
	play_audio = 1
	pause_after_audio = 0
}
else {
	play_audio = 0
}
__end__
set _prepare ""




C_repeat_if

___run__
// HACK:
// Strip cursor_roi:

cursor_roi = removeSubstring(cursor_roi, ";")


// Determine whether or not to repeat the current trial (cycle):
// By default, the trial is repeated
repeat_cycle_variable = 1
// Except when the continue button is clicked AND the child
// selected an image:
if (cursor_roi === "continue_arrow") {
	repeat_cycle_variable = 0
	}
__end__
